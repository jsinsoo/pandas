01. 반별 평균 계산하기

import pandas as pd

df = pd.DataFrame({
    "반": ["A", "A", "B", "B"],
    "이름": ["민수", "지아", "하준", "서연"],
    "점수": [90, 80, 70, 85]
})

result = df.groupby("반")["점수"].mean()
