10. CSV 읽기 → 처리 → 저장 흐름

import pandas as pd
from io import StringIO

csv_text = '''이름,점수
민수,90
지아,75
하준,82
'''

df = pd.read_csv(StringIO(csv_text))
df["합격여부"] = df["점수"].apply(lambda x: "합격" if x >= 80 else "재도전")
result_csv = df.to_csv(index=False)

print(df["합격여부"].tolist())
