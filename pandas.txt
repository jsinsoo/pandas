[문항 100] 예시 10. 분석 결과 저장까지
------------------------------------------------------------

import pandas as pd
df = pd.DataFrame({'반':['1반','1반','2반','2반'], '점수':[90,80,95,85]})
summary = df.groupby('반')['점수'].agg(['mean', 'max', 'min']).reset_index()
summary.to_csv('summary.csv', index=False, encoding='utf-8-sig')
print(summary)

[정답]
        반  mean  max  min
0  1반  85.0   90   80
1  2반  90.0   95   85
