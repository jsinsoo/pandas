06. 두 조건을 모두 만족하는 학생 찾기

import pandas as pd

df = pd.DataFrame({
    "이름": ["민수", "지아", "하준", "서연"],
    "국어": [90, 85, 78, 95],
    "수학": [88, 95, 80, 70]
})

result = df[(df["국어"] >= 85) & (df["수학"] >= 85)]

07. query로 상위권 학생 찾기

import pandas as pd

df = pd.DataFrame({
    "이름": ["민수", "지아", "하준", "서연"],

})

result = df.query("점수 >= 90")
