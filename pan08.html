10. 월별 매출과 이동평균 함께 분석

import pandas as pd

df = pd.DataFrame({
    "날짜": pd.date_range("2026-01-01", periods=6, freq="ME"),

})

df["누적매출"] = df["매출"].cumsum()
df["이동평균"] = df["매출"].rolling(window=3).mean()
