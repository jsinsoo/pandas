05. 전처리 함수를 파이프라인으로 연결

import pandas as pd

def add_average(df):
    df["평균"] = df[["국어", "수학"]].mean(axis=1)
    return df

df = pd.DataFrame({
    "이름": ["민수", "지아"],
    "국어": [90, 85],
    "수학": [88, 95]
})

result = df.pipe(add_average)
print(result["평균"].tolist())

06. 메서드 체이닝으로 매출 요약

import pandas as pd

df = pd.DataFrame({
    "지역": ["전주 ", "전주", "군산"],
    "매출": [100, 150, 80]
})

result = (
    df.assign(지역=df["지역"].str.strip())
      .groupby("지역")["매출"]
      .sum()
      .reset_index()
)

print(result.to_dict("records"))

10. 고급 문법을 활용한 자동 분석 흐름

import pandas as pd

df = pd.DataFrame({
    "학생": ["민수", "지아", "하준", "서연"],
    "점수": [95, 82, 70, 88],
    "반": ["A", "A", "B", "B"]
})

result = (
    df.assign(등급=df["점수"].apply(lambda x: "우수" if x >= 85 else "보통"))
      .groupby(["반", "등급"])["학생"]
      .count()
      .unstack(fill_value=0)
)

print(result)
