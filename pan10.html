01. 학생 성적 막대그래프 프로젝트

import pandas as pd
import matplotlib.pyplot as plt

df = pd.DataFrame({
    "이름": ["민수", "지아", "하준"],
    "점수": [90, 85, 78]
})

print(df["점수"].mean())
df.plot(kind="bar", x="이름", y="점수", legend=False)
plt.title("학생 성적 분석")
plt.ylabel("점수")
plt.tight_layout()
plt.show()

02. 매출 대시보드형 선그래프

import pandas as pd
import matplotlib.pyplot as plt

df = pd.DataFrame({
    "월": ["1월", "2월", "3월", "4월"],
    "매출": [100, 150, 130, 180]
})

df["누적매출"] = df["매출"].cumsum()
print(df["누적매출"].tolist())

df.plot(x="월", y=["매출", "누적매출"], marker="o")
plt.title("월별 매출 대시보드")
plt.tight_layout()
plt.show()

03. 카테고리별 평균 점수 분석

import pandas as pd

df = pd.DataFrame({
    "반": ["A", "A", "B", "B"],
    "점수": [90, 80, 70, 85]
})

summary = df.groupby("반")["점수"].mean().reset_index()
print(summary)

try:
    import seaborn as sns
    import matplotlib.pyplot as plt
    sns.barplot(data=summary, x="반", y="점수")
    plt.title("반별 평균 점수")
    plt.tight_layout()
    plt.show()
except ImportError:
    print("seaborn이 설치되어 있지 않습니다.")

    "국어": [90, 85, 78, 95],
    "수학": [88, 95, 80, 70]

    df.assign(평균=lambda x: (x["국어"] + x["수학"]) / 2)
      .assign(등급=lambda x: x["평균"].apply(lambda v: "우수" if v >= 85 else "보통"))
      .sort_values("평균", ascending=False)
)

print(result[["이름", "평균", "등급"]].head(2))
