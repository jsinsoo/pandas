01. 학생 정보와 성적 결합

import pandas as pd

students = pd.DataFrame({
    "학번": [101, 102, 103],
    "이름": ["민수", "지아", "하준"]
})

scores = pd.DataFrame({
    "학번": [101, 102, 103],
    "점수": [90, 85, 78]
})

result = pd.merge(students, scores, on="학번")
print(result.loc[1, "점수"])
